## 步骤

- [ ] 重写 `cmd/server/main.go`

```go
package main

import (
    "log"
    "strings"
    "github.com/gin-gonic/gin"
    "github.com/wikiglobal/suanming-agent/internal/orchestrator"
    "github.com/wikiglobal/suanming-agent/internal/sse"
    "github.com/wikiglobal/suanming-agent/internal/tools"
)

func main() {
    reg := tools.NewRegistry()
    reg.Register(&tools.BaziCalcTool{})

    orch := orchestrator.New(reg)

    r := gin.Default()
    r.Use(func(c *gin.Context) {
        c.Header("Access-Control-Allow-Origin", "*")
        c.Header("Access-Control-Allow-Methods", "POST,GET,OPTIONS")
        c.Header("Access-Control-Allow-Headers", "Content-Type")
        if c.Request.Method == "OPTIONS" { c.AbortWithStatus(204); return }
        c.Next()
    })

    r.GET("/api/health", func(c *gin.Context) { c.JSON(200, gin.H{"status":"ok"}) })

    r.POST("/api/chat", func(c *gin.Context) {
        var req struct {
            Message   string `json:"message"`
            SessionID string `json:"session_id"`
        }
        if err := c.BindJSON(&req); err != nil {
            c.JSON(400, gin.H{"error":"bad request"})
            return
        }
        if strings.TrimSpace(req.Message) == "" {
            c.JSON(400, gin.H{"error":"message is required"})
            return
        }
        if req.SessionID == "" { req.SessionID = "default" }
        orch.Run(sse.NewWriter(c), req.SessionID, req.Message)
    })

    log.Println("命理大师 :8080")
    r.Run(":8080")
}
```

- [ ] `go build ./cmd/server/` 编译成功

- [ ] curl 测试

```bash
# 场景1: 信息不全
curl -N -X POST http://localhost:8080/api/chat \
  -H "Content-Type: application/json" \
  -d '{"message":"帮我算八字"}'
# 预期: event:thinking → event:text (追问) → event:done

# 场景2: 完整排盘
curl -N -X POST http://localhost:8080/api/chat \
  -H "Content-Type: application/json" \
  -d '{"message":"1990年5月20日早上8点，男","session_id":"test1"}'
# 预期: event:thinking → event:tool_call → event:component(bazi-chart) → event:done

# 场景3: 追问复用
curl -N -X POST http://localhost:8080/api/chat \
  -H "Content-Type: application/json" \
  -d '{"message":"今年运势怎么样","session_id":"test1"}'
# 预期: event:thinking("复用已有命盘") → event:text → event:done

# 场景4: 缺少 message
curl -X POST http://localhost:8080/api/chat \
  -H "Content-Type: application/json" \
  -d '{"session_id":"test1"}'
# 预期: HTTP 400 {"error":"message is required"}
```
