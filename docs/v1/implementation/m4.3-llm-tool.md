```go
package orchestrator

import (
    "encoding/json"
    "fmt"

    "github.com/wikiglobal/suanming-agent/internal/mcp"
    "github.com/wikiglobal/suanming-agent/internal/state"
)

func buildKnowledgeQuery(st *state.SessionState) string {
    question := currentQuestion(st)
    return fmt.Sprintf(
        "命盘：%v；问题：%s；请检索与此问题最相关的命理规则、典籍原文和解释。",
        st.BaziResult,
        question,
    )
}

func currentQuestion(st *state.SessionState) string {
    if st.LastUserQuestion != "" {
        return st.LastUserQuestion
    }
    return "请先给出一段简明的命盘总评。"
}

func buildInterpretPrompt(st *state.SessionState, passages []mcp.Passage) string {
    tpl, err := os.ReadFile("prompts/interpret.md")
    if err != nil {
        tpl = []byte("你是精通八字命理的中文咨询师。请基于给定命盘和问题作答，不要暴露推理过程。")
    }
    profileJSON, _ := json.Marshal(st.Profile)
    baziJSON, _ := json.Marshal(st.BaziResult)
    passagesJSON, _ := json.Marshal(passages)

    return string(tpl) + `

## 运行时上下文

### 出生资料
` + string(profileJSON) + `

### 命盘结果
` + string(baziJSON) + `

### 知识引用
` + string(passagesJSON) + `

### 当前问题
` + currentQuestion(st)
}
```

- [ ] 在 `internal/orchestrator/orchestrator.go` 添加知识检索和流式生成

```go
func (o *Orchestrator) runKnowledgeSearch(sw *sse.Writer, st *state.SessionState) []mcp.Passage {
    tool, ok := o.tools.Get("knowledge_search")
    if !ok {
        sw.Send("thinking", map[string]any{
            "agent":"orchestrator","text":"知识检索未注册，跳过引用检索。",
        })
        return []mcp.Passage{}
    }

    query := buildKnowledgeQuery(st)
    sw.Send("tool_call", map[string]any{
        "tool":"knowledge_search",
        "params":map[string]any{"query": query, "topK": 3},
    })

    result, err := tool.Execute(map[string]any{"query": query, "topK": 3})
    if err != nil {
        sw.Send("thinking", map[string]any{
            "agent":"orchestrator","text":"知识检索失败，继续直接解读命盘。",
        })
        return []mcp.Passage{}
    }

    payload, ok := result.(map[string]any)
    if !ok {
        return []mcp.Passage{}
    }
    passages, _ := payload["passages"].([]mcp.Passage)
    if len(passages) > 0 {
        sw.Send("component", map[string]any{
            "type":"knowledge-sources",
            "payload":passages,
        })
    }
    return passages
}

func (o *Orchestrator) streamInterpretation(sw *sse.Writer, st *state.SessionState, passages []mcp.Passage) error {
    client := llm.NewClient()
    systemPrompt := buildInterpretPrompt(st, passages)
    messages := []llm.Message{
        {Role: "user", Content: currentQuestion(st)},
    }

    return client.ChatStream(systemPrompt, messages, func(chunk string) {
        sw.Send("text", map[string]any{"content": chunk})
    })
}
```

- [ ] 为 `orchestrator.go` 补充 import: `internal/llm`, `internal/mcp`

```go
passages := o.runKnowledgeSearch(sw, st)
if err := o.streamInterpretation(sw, st, passages); err != nil {
```

## 降级约定

- `knowledge_search` 失败：发 `thinking`，继续调用 LLM
- DeepSeek API 失败：发 `error + done`
- `LLM_API_KEY` 未设置：启动时报错退出
