## 步骤

- [ ] 创建 `internal/tools/bazi_calc_test.go`

```go
package tools

import "testing"

func TestBaziCalcKnownCase(t *testing.T) {
    tool := &BaziCalcTool{}
    result, err := tool.Execute(map[string]any{
        "year": float64(1990), "month": float64(5),
        "day": float64(20), "hour": float64(8), "gender": "男",
    })
    if err != nil { t.Fatalf("Execute: %v", err) }
    data := result.(map[string]any)

    pillars := data["pillars"].([]map[string]string)
    if len(pillars) != 4 { t.Errorf("pillars=%d, want 4", len(pillars)) }
    if data["dayGan"].(string) == "" { t.Error("dayGan empty") }

    // 验证十神
    for _, p := range pillars {
        if shiShen, ok := p["shiShen"]; !ok || shiShen == "" {
            t.Errorf("pillar %s shiShen empty", p["name"])
        }
    }

    wuxing := data["wuxing"].(map[string]int)
    total := 0
    for _, v := range wuxing { total += v }
    if total != 8 { t.Errorf("wuxing total=%d, want 8", total) }

    dayun := data["dayun"].([]map[string]any)
    if len(dayun) != 8 { t.Errorf("dayun=%d, want 8", len(dayun)) }

    t.Logf("日主=%s 五行=%v 大运=%d步", data["dayGan"], wuxing, len(dayun))
}

func TestBaziCalcInvalidYear(t *testing.T) {
    tool := &BaziCalcTool{}
    _, err := tool.Execute(map[string]any{
        "year": float64(1800), "month": float64(1),
        "day": float64(1), "hour": float64(0), "gender": "男",
    })
    if err == nil { t.Error("expected error for year 1800") }
}
```

- [ ] 运行测试

```bash
go test ./internal/tools/ -v -run TestBaziCalc
```
